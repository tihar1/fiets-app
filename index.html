<script>

function bereken() {

    const watt = document.getElementById('wattage').value;

    const min = document.getElementById('tijd').value;

    

    if(watt > 0 && min > 0) {

        // Formule: Watt * (tijd in uren) * 3.6

        const kcal = Math.round(watt * (min / 60) * 3.6);

        document.getElementById('kcal').innerText = kcal;

        

        let suggestie = "";

        if(kcal < 400) {

            suggestie = "Suggestie: Een banaan of een bakje magere kwark. Licht herstel nodig.";

        } else if(kcal < 800) {

            suggestie = "Suggestie: Bord havermout met fruit of twee volkoren boterhammen met kipfilet.";

        } else {

            suggestie = "Suggestie: Grote maaltijd! Pasta met veel groenten en eiwitten (kip/tofu) + extra hydratatie.";

        }

        

        document.getElementById('suggestie').innerText = suggestie;

        document.getElementById('resultaat').style.display = 'block';

    } else {

        alert("Vul aub geldige getallen in.");

    }

}

</script>
